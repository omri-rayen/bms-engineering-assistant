\newcommand{\chapterseparatorpage}[1]{%
  \thispagestyle{empty}%
  \vspace*{0.10\textheight}%
  \begin{center}
    {\fontsize{15}{19}\selectfont\bfseries\color{chapterink} Chapitre \thechapter\par}
    \vspace{0.5em}
    {\color{chapterrule}\rule{0.14\textwidth}{0.8pt}\par}
    \vspace{1em}
    {\fontsize{24}{30}\selectfont\bfseries\color{chapterink} #1\par}
  \end{center}
  \vspace{1.6cm}%
  \begin{center}
    \begin{minipage}{0.78\textwidth}
      \begin{center}
        {\bfseries\large\color{chapterink} Plan du chapitre\par}
        \vspace{0.5em}%
        {\color{chapterrule}\rule{0.32\textwidth}{0.6pt}\par}
      \end{center}
      \vspace{0.8em}%
      \chapterplanbox
    \end{minipage}
  \end{center}
  \vfill
}

\newcommand{\chapteropeningheading}[1]{%
  \thispagestyle{plain}%
  \vspace*{2.4cm}%
  \begin{center}
    {\fontsize{16}{20}\selectfont\bfseries\color{chapterink} Chapitre \thechapter\par}
    \vspace{0.6em}%
    {\color{chapterrule}\rule{0.16\textwidth}{0.8pt}\par}
    \vspace{1em}%
    {\fontsize{26}{32}\selectfont\bfseries\color{chapterink} #1\par}
  \end{center}
  \vspace{1.5cm}%
  \@afterindentfalse\@afterheading
}

\newcommand{\reportchapter}[1]{%
  \cleardoublepage
  \begingroup
    \def\@makechapterhead##1{}%
    \@chapter[{#1}]{#1}%
  \endgroup
  \chapterseparatorpage{#1}%
  \clearpage
  \chapteropeningheading{#1}%
}
\makeatother
